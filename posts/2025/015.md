<script setup>
import { ref, onMounted } from 'vue'

// Windows 版本
const winUrl = ref('')
const winVersion = ref('')
const winUpdateTime = ref('')
const winLoading = ref(true)

// Android 版本
const androidUrl = ref('')
const androidVersion = ref('')
const androidUpdateTime = ref('')
const androidLoading = ref(true)

onMounted(async () => {
  // 并行请求 Windows 和 Android 版本
  const fetchWin = fetch('https://www.qiyou.cn/api/front/get_last_client', {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' }
  }).then(res => res.json()).then(data => {
    winUrl.value = data.client_url
    winVersion.value = data.client_version
    winUpdateTime.value = data.add_time
  }).catch(e => console.error('获取Windows版本失败:', e))
    .finally(() => winLoading.value = false)

  const fetchAndroid = fetch('https://www.qiyou.cn/api/front/get_last_client', {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: 'client_type_node=20&channel_id=1120'
  }).then(res => res.json()).then(data => {
    androidUrl.value = data.client_url
    androidVersion.value = data.client_version
    // 从链接提取时间戳并转换 (格式: /1764319892845/)
    const match = data.client_url.match(/\/(\d{13})\//)
    if (match) {
      const ts = parseInt(match[1])
      const date = new Date(ts)
      androidUpdateTime.value = `${date.getFullYear()}-${String(date.getMonth()+1).padStart(2,'0')}-${String(date.getDate()).padStart(2,'0')} ${String(date.getHours()).padStart(2,'0')}:${String(date.getMinutes()).padStart(2,'0')}`
    }
  }).catch(e => console.error('获取Android版本失败:', e))
    .finally(() => androidLoading.value = false)

  await Promise.all([fetchWin, fetchAndroid])
})

const copyLink = (link, type) => {
  navigator.clipboard.writeText(link).then(() => {
    alert('链接已复制，关注无度谢谢喵~')
  })
  if (window.umami) {
    window.umami.track('奇游复制', { type, link })
  }
}

const trackDownload = (type, url, version) => {
  if (window.umami) {
    window.umami.track('奇游下载', { type, url, version })
  }
}
</script>
