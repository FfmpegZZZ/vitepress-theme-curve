import { ref, onMounted } from 'vue'

// Windows 版本
const winUrl = ref('')
const winVersion = ref('')
const winUpdateTime = ref('')
const winLoading = ref(true)

// Android 版本
const androidUrl = ref('')
const androidVersion = ref('')
const androidUpdateTime = ref('')
const androidLoading = ref(true)

// Mac 版本
const macUrl = ref('')
const macVersion = ref('')
const macLoading = ref(true)

// XSS 防护：验证 URL 是否为安全的下载链接
const isValidUrl = (url) => {
  if (!url || typeof url !== 'string') return false
  try {
    const parsed = new URL(url)
    // 只允许 https 协议和白名单域名
    const allowedDomains = ['qiyou.cn', 'static.qiyou.cn', 'apifast.qiyou.cn', 'download.qiyou.cn', 'qijihezi.cn', 'coolsvcoss.qijihezi.cn']
    const isHttps = parsed.protocol === 'https:'
    const isDomainAllowed = allowedDomains.some(d => parsed.hostname === d || parsed.hostname.endsWith('.' + d))
    return isHttps && isDomainAllowed
  } catch {
    return false
  }
}

// XSS 防护：清理版本号，只保留安全字符
const sanitizeVersion = (version) => {
  if (!version || typeof version !== 'string') return ''
  // 只允许数字、点、字母和短横线
  return version.replace(/[^a-zA-Z0-9.\-]/g, '').slice(0, 20)
}

// XSS 防护：清理时间字符串
const sanitizeTime = (time) => {
  if (!time || typeof time !== 'string') return ''
  // 只允许日期时间格式的字符
  return time.replace(/[^0-9\-: ]/g, '').slice(0, 20)
}

onMounted(async () => {
  // 并行请求 Windows、Android 和 Mac 版本
  const fetchWin = fetch('https://www.qiyou.cn/api/front/get_last_client', {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' }
  }).then(res => res.json()).then(data => {
    if (isValidUrl(data.client_url)) {
      winUrl.value = data.client_url
    }
    winVersion.value = sanitizeVersion(data.client_version)
    winUpdateTime.value = sanitizeTime(data.add_time)
  }).catch(e => console.error('获取Windows版本失败:', e))
    .finally(() => winLoading.value = false)

  const fetchAndroid = fetch('https://www.qiyou.cn/api/front/get_last_client', {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: 'client_type_node=20&channel_id=1120'
  }).then(res => res.json()).then(data => {
    if (isValidUrl(data.client_url)) {
      androidUrl.value = data.client_url
      // 从链接提取时间戳并转换 (格式: /1764319892845/)
      const match = data.client_url.match(/\/(\d{13})\//)
      if (match) {
        const ts = parseInt(match[1])
        const date = new Date(ts)
        androidUpdateTime.value = `${date.getFullYear()}-${String(date.getMonth()+1).padStart(2,'0')}-${String(date.getDate()).padStart(2,'0')} ${String(date.getHours()).padStart(2,'0')}:${String(date.getMinutes()).padStart(2,'0')}`
      }
    }
    androidVersion.value = sanitizeVersion(data.client_version)
  }).catch(e => console.error('获取Android版本失败:', e))
    .finally(() => androidLoading.value = false)

  const fetchMac = fetch('https://apifast.qiyou.cn/api/common_bll/v1/official_web/download_url?client_type=MAC', {
    method: 'GET',
    headers: {
      'Content-Type': 'application/x-www-form-urlencoded; charset=UTF-8',
      'Accept': 'application/json, text/plain, */*'
    }
  }).then(res => res.json()).then(data => {
    if (data.download_url && isValidUrl(data.download_url)) {
      macUrl.value = data.download_url
      // 从链接中提取版本号 (格式: MacGameAccelerator-vrelease-131-Release)
      const match = data.download_url.match(/vrelease-(\d+)/)
      if (match) {
        macVersion.value = sanitizeVersion(match[1])
      }
    }
  }).catch(e => console.error('获取Mac版本失败:', e))
    .finally(() => macLoading.value = false)

  await Promise.all([fetchWin, fetchAndroid, fetchMac])
})

const copyLink = (link, type) => {
  navigator.clipboard.writeText(link).then(() => {
    if (typeof $message !== 'undefined') {
      $message.success('链接已复制，关注无度谢谢喵~')
    }
  }).catch(() => {
    if (typeof $message !== 'undefined') {
      $message.error('复制失败，请手动复制')
    }
  })
  if (window.umami) {
    window.umami.track('奇游复制', { type, link })
  }
}

const trackDownload = (type, url, version) => {
  if (window.umami) {
    window.umami.track('奇游下载', { type, url, version })
  }
}
</script>
